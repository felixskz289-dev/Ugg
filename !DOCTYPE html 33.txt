let zhVoice = null;
function pickVoice(){
  const vs = speechSynthesis.getVoices();
  zhVoice = vs.find(v=>v.lang && v.lang.toLowerCase().startsWith('zh') && /CN|china/i.test(v.lang))
         || vs.find(v=>v.lang && v.lang.toLowerCase().startsWith('zh')) || null;
}
if('speechSynthesis' in window){ pickVoice(); speechSynthesis.onvoiceschanged = pickVoice; }
function speak(text, rate=0.75){
  if(!('speechSynthesis' in window)) return;
  speechSynthesis.cancel();
  const u = new SpeechSynthesisUtterance(text);
  u.lang='zh-CN'; if(zhVoice) u.voice=zhVoice; u.rate=rate;
  speechSynthesis.speak(u);
}

let sIdx=0, selTokens=[];
function renderSentence(){
  if(sIdx>=SENTENCES.length){ sentenceBox.innerHTML='<div style="text-align:center;font-size:24px">🌟 Молодцы! Все предложения собраны!</div>'; return; }
  const S=SENTENCES[sIdx]; selTokens=[];
  const shuffled=[...S.tokens].sort(()=>Math.random()-.5);
  sentenceBox.innerHTML=`
    <div style="text-align:center;font-size:20px;background:var(--yellow);border-radius:14px;padding:12px;margin-bottom:6px">🇷🇺 ${S.rus}</div>
    <div class="answer-bar" id="ansBar"></div>
    <div class="tokens">${shuffled.map((t,i)=>`<div class="tok" id="tok${i}" onclick="pickTok(${i},'${t}')">${t}</div>`).join('')}</div>
    <div style="text-align:center"><button class="btn pink" onclick="checkSentence()">✅ Проверить</button>
    <button class="btn ghost" onclick="clearSentence()">↺ Сброс</button></div>
    <div class="feedback" id="sFeed"></div>`;
}
function pickTok(i,t){
  document.getElementById('tok'+i).classList.add('used');
  selTokens.push(t);
  const chip=document.createElement('div'); chip.className='ans-chip'; chip.textContent=t;
  chip.onclick=()=>{ chip.remove(); selTokens.splice(selTokens.indexOf(t),1); document.getElementById('tok'+i).classList.remove('used'); };
  document.getElementById('ansBar').appendChild(chip);
}
function clearSentence(){ renderSentence(); }
function checkSentence(){
  const S=SENTENCES[sIdx];
  if(selTokens.join('')===S.tokens.join('')){
    sFeed.innerHTML='✅ 对! Правильно!'; sFeed.style.color='var(--green2)';
    speak(selTokens.join(''));
    confetti(15);
    setTimeout(()=>{sIdx++; renderSentence();},1800);
  } else { sFeed.innerHTML='❌ Пока нет, попробуй ещё!'; sFeed.style.color='#ff7b7b'; }
}
renderSentence();

  <div class="feedback" id="mFeed" style="text-align:center"></div>`;
  matchBox.querySelectorAll('[data-side="h"] .m-item, .m-item').forEach(el=>{
    const isH=el.dataset.side==='h';
  });
}
function mPick(el){
  if(el.classList.contains('done')) return;
  if(matchSel===el){ el.classList.remove('sel'); matchSel=null; return; }
  el.classList.add('sel');
  if(matchSel){
    if(matchSel.dataset.side!==el.dataset.side && matchSel.dataset.key===el.dataset.key){
      matchSel.classList.add('done'); el.classList.add('done'); matchDone++;
      mFeed.innerHTML='✅ Пара!'; mFeed.style.color='var(--green2)';
      speak(el.dataset.key);
      matchSel=null;
      if(matchDone===CITY_PAIRS.length){ mFeed.innerHTML='🎉 Все пары собраны! Дайте обеим командам по 2 очка!'; confetti(15); }
    } else {
      const a=matchSel,b=el; matchSel=null;
      a.classList.add('shake'); b.classList.add('shake'); mFeed.innerHTML='❌ Мимо!'; mFeed.style.color='#ff7b7b';
      setTimeout(()=>{a.classList.remove('sel','shake');b.classList.remove('sel','shake');},450);
    }
  } else matchSel=el;
}

let buzzQueue=[], curBuzz=null, buzzed=null;
function nextBuzz(){
  if(!buzzQueue.length) buzzQueue=[...BUZZ_POOL].sort(()=>Math.random()-.5);
  curBuzz=buzzQueue.pop(); buzzed=null;
  buzzWord.textContent='🇷🇺 '+curBuzz.t;
  buzzMsg.textContent='';
}
function buzz(t){
  if(!curBuzz || buzzed) return; buzzed=t;
  buzzWord.innerHTML=`🐼🐉 Слово: <b>${curBuzz.t}</b> — команда ${t===1?'🐼 Панд':'🐉 Драконов'}, ваш ответ!`;
  speak(curBuzz.h);
}
function buzzCorrect(t){
  if(!curBuzz) return;
  if(t===buzzed){ addP(t,2); buzzMsg.textContent='⚡ +2 очка команде '+(t===1?'🐼 Панд':'🐉 Драконов')+'!'; confetti(12); }
  else { addP(t===1?2:1,2); buzzMsg.textContent='❌ Ошибка! +2 очка сопернику!'; }
  curBuzz=null; buzzed=null; buzzWord.textContent='';
}
